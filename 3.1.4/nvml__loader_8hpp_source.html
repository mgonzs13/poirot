55// ABI layout locks: the driver writes these structs directly. The v2 list API
56// is used because drivers in the 535.43-535.98 window grew the v3 struct,
57// which would overrun a 24-byte buffer.
58static_assert(sizeof(NvmlUtilization) == 8, "NVML utilization layout changed");
59static_assert(sizeof(NvmlMemory) == 24, "NVML memory layout changed");
60static_assert(sizeof(NvmlProcessUtilizationSample) == 32,
61 "NVML process utilization layout changed");
62static_assert(sizeof(NvmlProcessInfoV2) == 24,
63 "NVML process info layout changed");
